        function updateChengduTime() {
            const now = new Date();

            const options = { timeZone: 'Asia/Shanghai', hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' };

        function setEmotion(type) {
            // 更新按钮激活状态
            const btns = document.querySelectorAll('.emotion-btn');
            btns.forEach(btn => {
                btn.classList.toggle('active', btn.getAttribute('onclick').includes(type));
            });

            const config = emotions[type];
            const img = document.getElementById('barbieImg');
            const bubble = document.getElementById('emotionBubble');
            const fx = document.getElementById('fxLayer');
            const stage = document.getElementById('stage');

            // 联动 1：更新提示文本与气泡
            bubble.innerText = config.bubble;

            // 联动 2：更新视觉滤镜与色彩
            img.style.filter = config.filter;

            // 联动 3：更新背景光效与动效
            fx.style.background = config.bgFx;

            // 联动 4：震颤/爆炸物理动作
            stage.className = "character-stage " + config.anim;
        }
